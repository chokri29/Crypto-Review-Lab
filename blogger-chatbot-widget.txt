<script>
  (function() {
    var container = document.getElementById('crypto-chatbot-root');
    if (!container) return;
    
    // Create the widget iframe
    var iframe = document.createElement('iframe');
    iframe.id = 'crypto-chatbot-iframe';
    iframe.frameBorder = '0';
    iframe.allowTransparency = 'true';
    
    // Explicitly target your dynamic deployment URL
    var deploymentUrl = "https://ais-pre-pdkemchly6453zkdn6oggx-534731101661.europe-west1.run.app";
    iframe.src = deploymentUrl + "/widget/chat";
    container.appendChild(iframe);
    
    // Create floating launcher button
    var launcher = document.createElement('button');
    launcher.id = 'crypto-chatbot-launcher';
    launcher.setAttribute('aria-label', 'Chat with Lab Auditor');
    
    // SVG Speech Bubble icon
    launcher.innerHTML = '<svg fill="none" stroke="currentColor" stroke-linecap="round" stroke-linejoin="round" stroke-width="2" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path d="M21 15a2 2 0 0 1-2 2H7l-4 4V5a2 2 0 0 1 2-2h14a2 2 0 0 1 2 2z"></path></svg>';
    container.appendChild(launcher);
    
    var isOpen = false;
    launcher.onclick = function() {
      isOpen = !isOpen;
      if (isOpen) {
        iframe.classList.add('widget-visible');
        launcher.classList.add('launcher-open');
        // Set Close (X) icon
        launcher.innerHTML = '<svg fill="none" stroke="currentColor" stroke-linecap="round" stroke-linejoin="round" stroke-width="2" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><line x1="18" y1="6" x2="6" y2="18"></line><line x1="6" y1="6" x2="18" y2="18"></line></svg>';
        launcher.style.transform = "rotate(90deg)";
      } else {
        iframe.classList.remove('widget-visible');
        launcher.classList.remove('launcher-open');
        // Restore speech bubble icon
        launcher.innerHTML = '<svg fill="none" stroke="currentColor" stroke-linecap="round" stroke-linejoin="round" stroke-width="2" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path d="M21 15a2 2 0 0 1-2 2H7l-4 4V5a2 2 0 0 1 2-2h14a2 2 0 0 1 2 2z"></path></svg>';
        launcher.style.transform = "rotate(0deg)";
      }
    };
  })();
</script>
